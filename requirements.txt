# Pharma Study Analyser - Python dependencies
# Install: python -m pip install -r requirements.txt
# Versions are not pinned. After a working install, run `pip freeze > requirements.lock.txt` to lock them.

# ---- LLM framework ----
langchain
langchain-core
langchain-community
langchain-openai          # ChatOpenAI, AzureChatOpenAI, OpenAIEmbeddings
langchain-text-splitters  # recursive and parent-child chunking
langchain-azure-ai        # Azure AI Search vector store integration
langgraph                 # agents: map-reduce insights, verdict and slide flows
openai
tiktoken                  # token counting for usage and cost tracking

azure-storage-blob        # PDF storage in Blob

azure-keyvault-secrets    # secrets from Key Vault
azure-ai-documentintelligence  # layout and table extraction (optional)

sqlalchemy

pypdf

pydantic                  # structured extraction schema and validators
pydantic-settings
pandas
numpy

python-dotenv             # load .env

fastapi

plotly                    # forest plots, usage dashboard
